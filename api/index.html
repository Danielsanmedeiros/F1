    def settings_from_env(env: Mapping[str, str]) -> Settings:
    """Monta as configurações a partir das variáveis informadas.

    Args:
        env: Variáveis de ambiente.

    Returns:
        Configurações validadas, com os valores padrão nas variáveis ausentes.

    Raises:
        ValueError: Se o intervalo entre chamadas não for um número não negativo.
    """
    raw_interval = env.get(
        "OPENF1_REQUEST_INTERVAL_SECONDS", str(DEFAULT_REQUEST_INTERVAL_SECONDS)
    )
    try:
        interval = float(raw_interval)
    except ValueError as exc:
        raise ValueError(
            f"OPENF1_REQUEST_INTERVAL_SECONDS inválido: {raw_interval!r}"
        ) from exc
    if interval < 0:
        raise ValueError("OPENF1_REQUEST_INTERVAL_SECONDS não pode ser negativo")

    return Settings(
        base_url=env.get("OPENF1_BASE_URL", DEFAULT_BASE_URL),
        request_interval_seconds=interval,
        data_dir=Path(env.get("F1_DATA_DIR", DEFAULT_DATA_DIR)),
        log_level=env.get("LOG_LEVEL", DEFAULT_LOG_LEVEL).upper(),
        data_snapshot_url=env.get("F1_DATA_SNAPSHOT_URL") or None,
    )

    class OpenF1Client:
    """Consulta a API com intervalo entre chamadas e novas tentativas.

    Args:
        base_url: URL base da API.
        request_interval: Intervalo mínimo entre chamadas, em segundos.
        max_retries: Quantidade de novas tentativas após a primeira chamada.
        backoff_base: Espera da primeira nova tentativa, em segundos. Dobra a cada
            tentativa, salvo quando a API informa `Retry-After`.
        timeout: Tempo limite de cada chamada, em segundos.
        transport: Transporte HTTP alternativo, usado nos testes.
        sleep: Função de espera.
        clock: Relógio monotônico.
    """

    def __init__(
        self,
        base_url: str,
        request_interval: float = 2.0,
        *,
        max_retries: int = 5,
        backoff_base: float = 2.0,
        timeout: float = 60.0,
        transport: httpx.BaseTransport | None = None,
        sleep: Callable[[float], None] = time.sleep,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self._http = httpx.Client(
            base_url=base_url, timeout=timeout, transport=transport
        )
        self._request_interval = request_interval
        self._max_retries = max_retries
        self._backoff_base = backoff_base
        self._sleep = sleep
        self._clock = clock
        self._last_request_at: float | None = None

    def __enter__(self) -> Self:
        return self

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc: BaseException | None,
        traceback: TracebackType | None,
    ) -> None:
        self.close()

    def close(self) -> None:
        """Encerra as conexões HTTP."""
        self._http.close()

    def get(self, endpoint: str, **params: str | int) -> list[dict[str, Any]]:
        """Busca os registros de um endpoint.

        Args:
            endpoint: Nome do endpoint, como `laps`.
            **params: Filtros da consulta, como `session_key=9165`.

        Returns:
            Registros retornados. A lista é vazia quando não há resultados.

        Raises:
            ApiAccessError: Se a API recusar o acesso (401 ou 403).
            ApiRequestError: Se a chamada falhar de forma definitiva ou se as novas
                tentativas se esgotarem.
        """
        attempt = 0
        while True:
            self._wait_for_interval()
            try:
                response = self._http.get(endpoint, params=params)
            except httpx.TransportError as exc:
                failure = f"erro de rede: {exc}"
                delay = self._backoff_delay(attempt)
            else:
                if response.status_code != 429 and response.status_code < 500:
                    return self._parse(response, endpoint)
                failure = f"HTTP {response.status_code}"
                delay = _retry_after_seconds(response) or self._backoff_delay(attempt)

            if attempt >= self._max_retries:
                raise ApiRequestError(
                    f"{endpoint} {params}: {failure} após {attempt + 1} tentativas"
                )
            attempt += 1
            logger.warning(
                "%s %s: %s. Nova tentativa %d de %d em %.1f s",
                endpoint,
                params,
                failure,
                attempt,
                self._max_retries,
                delay,
            )
            self._sleep(delay)

    def _wait_for_interval(self) -> None:
        if self._last_request_at is not None:
            remaining = self._request_interval - (self._clock() - self._last_request_at)
            if remaining > 0:
                self._sleep(remaining)
        self._last_request_at = self._clock()

    def _backoff_delay(self, attempt: int) -> float:
        return self._backoff_base * 2**attempt

    def _parse(self, response: httpx.Response, endpoint: str) -> list[dict[str, Any]]:
        status = response.status_code
        if status in (401, 403):
            raise ApiAccessError(f"HTTP {status}: {_error_detail(response)}")
        if status == 404 and "no results" in _error_detail(response).lower():
            return []
        if status != 200:
            raise ApiRequestError(
                f"{endpoint}: HTTP {status}: {_error_detail(response)}"
            )
        try:
            payload = response.json()
        except ValueError as exc:
            raise ApiRequestError(f"{endpoint}: resposta não é JSON") from exc
        if not isinstance(payload, list):
            raise ApiRequestError(f"{endpoint}: resposta não é uma lista")
        return payload

    def flatten_session_result(records: Sequence[Record]) -> list[Record]:

    def normalize_records(
    records: Sequence[Record], schema: EndpointSchema
) -> pl.DataFrame:
    """Converte registros da API em um DataFrame com as colunas do schema.

    Colunas opcionais ausentes são criadas com nulos. Colunas desconhecidas geram um
    aviso e são descartadas. Uma lista vazia gera um DataFrame vazio com o schema.

    Args:
        records: Registros retornados pela API.
        schema: Contrato do endpoint.

    Returns:
        DataFrame com as colunas e os tipos do schema.

    Raises:
        SchemaValidationError: Se faltar uma coluna obrigatória ou se algum valor
            tiver tipo inválido.
    """
    prepare = _PREPARERS.get(schema.name)
    prepared = prepare(records) if prepare else records
    if prepared:
        _check_columns(prepared, schema)
    return pl.DataFrame([_build_series(prepared, column) for column in schema.columns])

    def select_finished_sessions(sessions: pl.DataFrame, now: datetime) -> list[int]:
    """Seleciona as sessões encerradas e não canceladas.

        sessions: Tabela `sessions` normalizada.

    selected = sessions.filter(

    def enforce_quality(df: pl.DataFrame, schema: EndpointSchema) -> pl.DataFrame:
    """Aplica as regras de qualidade de um endpoint.

    Endpoints com chave não podem ter nulos nem duplicatas nela. Endpoints sem chave
    natural têm apenas as linhas idênticas removidas.

    Args:
        df: DataFrame normalizado.
        schema: Contrato do endpoint.

    Returns:
        DataFrame pronto para gravação.

    Raises:
        DataQualityError: Se a chave tiver nulos ou duplicatas.
    """
    if not schema.key:
        return df.unique(maintain_order=True)

    key = list(schema.key)
    null_rows = df.filter(pl.any_horizontal(pl.col(key).is_null())).height
    if null_rows:
        raise DataQualityError(
            f"{schema.name}: {null_rows} linhas com nulo na chave {key}"
        )

    duplicated_rows = df.height - df.select(key).unique().height
    if duplicated_rows:
        raise DataQualityError(
            f"{schema.name}: {duplicated_rows} linhas duplicadas na chave {key}"
        )
    return df

    def build_roster(drivers: pl.DataFrame, results: pl.DataFrame) -> pl.DataFrame:
    """Reúne os pilotos de uma sessão, na ordem do resultado.

    Pilotos sem posição ficam ao final, primeiro os que completaram mais voltas.

    Args:
        drivers: Tabela `drivers` da sessão.
        results: Tabela `session_result` da sessão.

    Returns:
        Uma linha por piloto, com sigla, equipe, cor, foto, estilo de linha, texto do
        resultado e `order`, que começa em 0. O segundo piloto de uma equipe recebe
        linha tracejada, para não depender só da cor.
    """
    merged = (
        drivers.select(_DRIVER_COLUMNS)
        .join(
            results.select(_RESULT_COLUMNS),
            on="driver_number",
            how="full",
            coalesce=True,
        )
        .sort(
            ["position", "number_of_laps", "driver_number"],
            descending=[False, True, False],
            nulls_last=True,
        )
    )

    rows: list[dict[str, Any]] = []
    teams_seen: set[str] = set()
    for order, row in enumerate(merged.iter_rows(named=True)):
        number = row["driver_number"]
        acronym = row["name_acronym"] or f"#{number}"
        team = row["team_name"] or "Equipe desconhecida"
        color = team_color(row["team_colour"])
        rows.append(
            {
                "order": order,
                "driver_number": number,
                "acronym": acronym,
                "full_name": row["full_name"] or acronym,
                "team_name": team,
                "color": color,
                "photo": driver_photo(row["headshot_url"], acronym, color),
                "line_dash": list(DASHED_LINE if team in teams_seen else SOLID_LINE),
                "position": row["position"],
                "number_of_laps": row["number_of_laps"],
                "points": row["points"],
                "result": result_text(row),
            }
        )
        teams_seen.add(team)
    return pl.DataFrame(
        rows,
        schema={
            "order": pl.Int64,
            "driver_number": pl.Int64,
            "acronym": pl.String,
            "full_name": pl.String,

            "position": pl.Int64,
            "number_of_laps": pl.Int64,

    )

    def format_duration(seconds: float) -> str:
    """Formata uma duração como `1:31:44.742` ou `1:32.608`.

    def positions_by_lap(
    positions: pl.DataFrame, laps: pl.DataFrame, roster: pl.DataFrame
) -> pl.DataFrame:
    """Calcula a posição de cada piloto ao final de cada volta.

    A API só registra as mudanças de posição. Cada mudança é atribuída à volta em
    andamento, e a posição é mantida até a última volta do piloto. A volta 0 é a
    posição antes da primeira volta registrada.

    Args:
        positions: Tabela `position` da sessão.
        laps: Tabela `laps` da sessão.
        roster: Resultado de `build_roster`.

    Returns:
        Uma linha por piloto e volta, com os dados de `roster`.
    """
    lookup = roster.select(_ROSTER_LOOKUP)
    if positions.is_empty():
        return pl.DataFrame(
            schema={"lap_number": pl.Int64, "position": pl.Int64, **lookup.schema}
        )

    lap_starts = (
        laps.filter(pl.col("date_start").is_not_null())
        .select("driver_number", "date_start", "lap_number")
        .sort("date_start")
    )
    events = (
        positions.select("driver_number", "date", "position")
        .sort("date")
        .join_asof(
            lap_starts,
            left_on="date",
            right_on="date_start",
            by="driver_number",
            strategy="backward",
            check_sortedness=False,
        )
        .with_columns(pl.col("lap_number").fill_null(0))
        .group_by("driver_number", "lap_number", maintain_order=True)
        .agg(pl.col("position").last())
    )
    last_laps = (
        pl.concat(
            [
                events.select("driver_number", "lap_number"),
                laps.select("driver_number", "lap_number"),
            ]
        )
        .group_by("driver_number")
        .agg(pl.col("lap_number").max().alias("last_lap"))
    )
    all_laps = last_laps.with_columns(
        pl.int_ranges(0, pl.col("last_lap") + 1, dtype=pl.Int64).alias("lap_number")
    ).explode("lap_number")
    return (
        all_laps.select("driver_number", "lap_number")
        .join(events, on=["driver_number", "lap_number"], how="left")
        .sort("driver_number", "lap_number")
        .with_columns(pl.col("position").forward_fill().over("driver_number"))
        .filter(pl.col("position").is_not_null())
        .join(lookup, on="driver_number")
    )

    def spread_labels(values: Sequence[float], min_gap: float) -> list[float]:
    """Afasta os rótulos de um eixo até que não se sobreponham.

    Rótulos próximos formam um grupo, centrado na média dos seus valores.

    Args:
        values: Posições originais, em ordem crescente.
        min_gap: Distância mínima entre rótulos, na unidade do eixo.

    Returns:
        Posições ajustadas, na mesma ordem.
    """
    groups: list[list[float]] = []
    for value in values:
        groups.append([value])
        while len(groups) > 1 and _groups_overlap(groups[-2], groups[-1], min_gap):
            last = groups.pop()
            groups[-1].extend(last)

    spread: list[float] = []
    for group in groups:
        center = sum(group) / len(group)
        spread.extend(
            center + (index - (len(group) - 1) / 2) * min_gap
            for index in range(len(group))
        )
    return spread
